\section{Presheaves, stalks and sheaves}
\label{section-presheaves-stalks-sheaves}
\label{section-presheaves}
\label{section-sheaves}
\label{section-stalks}

\begin{definition}
\label{definition-presheaf}
Let $X$ be a topological space and $\mathcal C$ a
category. A {\it presheaf} $\mathcal F$ on $X$
with values in $\mathcal C$ assigns an object
$\mathcal F(U)$ to each open $U\subset X$ and,
for every inclusion $V\subset U$, a restriction
morphism
$$
\operatorname{Res}_{U,V}:
\mathcal F(U)\longrightarrow\mathcal F(V),
$$
such that
$$
\operatorname{Res}_{U,U}=\operatorname{Id},
\qquad
\operatorname{Res}_{V,W}\circ
\operatorname{Res}_{U,V}
=
\operatorname{Res}_{U,W}
$$
whenever $W\subset V\subset U$.
Equivalently, a presheaf is a functor
$$
\operatorname{Open}(X)^{\mathrm{op}}
\longrightarrow \mathcal C.
$$
\end{definition}

\noindent
A morphism of presheaves is a natural
transformation.  When the target category is
concrete, elements of $\mathcal F(U)$ are called
{\it sections over $U$}.

\begin{example}[Constant presheaf]
\label{example-constant-presheaf}
For a fixed object $A\in\mathcal C$, the assignment
$\mathcal F(U)=A$ with identity restriction maps
defines the constant presheaf.
\end{example}

\begin{example}
\label{example-smooth-functions-presheaf}
If $X$ is a smooth manifold, then
$$
U\longmapsto C^\infty(U)
$$
with ordinary restriction of functions is a
presheaf.  The analogous constructions with
continuous or holomorphic functions will also be
used.
\end{example}

\begin{definition}
\label{definition-stalk}
For $p\in X$, the {\it stalk} of a presheaf
$\mathcal F$ at $p$ is
$$
\mathcal F_p
=
\varinjlim_{p\in U}\mathcal F(U),
$$
the filtered colimit over the open neighbourhoods
of $p$.
\end{definition}

\noindent
An element of $\mathcal F_p$ is a germ.  It is
represented by a pair $(U,s)$ with $p\in U$ and
$s\in\mathcal F(U)$; two such pairs define the
same germ exactly when the sections agree after
restriction to some smaller neighbourhood of $p$.

\begin{definition}
\label{definition-sheaf}
A presheaf $\mathcal F$ is a {\it sheaf} if local
sections glue uniquely.  Explicitly, if
$U=\bigcup_i U_i$ and $s_i\in\mathcal F(U_i)$
satisfy
$$
s_i|_{U_i\cap U_j}=s_j|_{U_i\cap U_j},
$$
then there exists a unique $s\in\mathcal F(U)$
whose restriction to every $U_i$ is $s_i$.
\end{definition}

\begin{example}
The presheaves of all functions, continuous
functions, smooth functions, and holomorphic
functions are sheaves.
\end{example}

\subsection{Sheafification}

\noindent
A presheaf need not satisfy the gluing axiom.  Its
{\it sheafification} is obtained by adding exactly
the locally represented sections that are missing.

For a presheaf $\mathcal F$, form the disjoint union
of its stalks
$$
\operatorname{Et}(\mathcal F)
=
\coprod_{p\in X}\mathcal F_p.
$$
For an open $U\subset X$, define
$\mathcal F^+(U)$ to be the functions
$s:U\to\operatorname{Et}(\mathcal F)$ such that
$s(p)\in\mathcal F_p$ and such that every
$p\in U$ has a neighbourhood on which $s$ is
represented by an honest section of $\mathcal F$.

\begin{proposition}
\label{ringed-spaces-proposition-sheafification}
The assignment $U\mapsto\mathcal F^+(U)$ is a
sheaf.  It is called the sheafification of
$\mathcal F$.
\end{proposition}

\begin{remark}
The set $\operatorname{Et}(\mathcal F)$ can be
topologized so that $\mathcal F^+(U)$ is precisely
the set of continuous sections over $U$ of the
projection
$\operatorname{Et}(\mathcal F)\to X$.
This is the {\it \'etale space} point of view.
\end{remark}

\begin{proposition}[Universal property of sheafification]
\label{ringed-spaces-proposition-sheafification-adjunction}
If $\mathcal F$ is a presheaf and $\mathcal G$ is
a sheaf on $X$, then
$$
\operatorname{Hom}(\mathcal F^+,\mathcal G)
\cong
\operatorname{Hom}(\mathcal F,\mathcal G^\circ),
$$
naturally, where $\mathcal G^\circ$ denotes the
underlying presheaf.
\end{proposition}

\subsection{Constant and skyscraper sheaves}

\noindent
The constant presheaf with value $M$ is usually
not a sheaf.  Its sheafification is the
{\it constant sheaf}, denoted $\underline M_X$.
Its sections are locally constant $M$-valued
functions.  In particular,
$$
(\underline M_X)_p\cong M
$$
for all $p\in X$.

\begin{definition}
\label{definition-skyscraper}
Fix $p\in X$ and an abelian group or vector space
$M$.  The {\it skyscraper sheaf} at $p$ with value
$M$ is
$$
\mathcal F(U)
=
\begin{cases}
M,&p\in U,\\
0,&p\notin U,
\end{cases}
$$
with the evident restriction maps.
\end{definition}

\begin{theorem}
\label{theorem-skyscraper-cohomology}
A skyscraper sheaf is flabby, hence acyclic for
global sections.
\end{theorem}

\section{Sheaves of vector spaces and exactness}
\label{ringed-spaces-section-sheaves-vector-spaces}
\label{section-kernels-cokernels}
\label{section-exact-sequences}

\noindent
For sheaves with values in an abelian category,
a morphism
$\varphi:\mathcal F\to\mathcal G$ induces maps
$\varphi_p:\mathcal F_p\to\mathcal G_p$ on all
stalks.  Injectivity, surjectivity and exactness
are tested stalkwise.

\begin{definition}
\label{definition-kernel-image-cokernel}
For a morphism
$\varphi:\mathcal F\to\mathcal G$ of sheaves of
abelian groups or modules, the kernel is computed
on open sets:
$$
(\ker\varphi)(U)
=
\ker\bigl(\mathcal F(U)\to\mathcal G(U)\bigr).
$$
The image and cokernel are obtained from the
corresponding presheaves by sheafification.
Equivalently,
$$
(\operatorname{coker}\varphi)_p
\cong
\operatorname{coker}
\bigl(\mathcal F_p\to\mathcal G_p\bigr).
$$
\end{definition}

\begin{remark}
\label{remark-image-cokernel}
This is why surjectivity of a sheaf morphism does
not mean surjectivity on sections over every open
set.  Surjectivity is a local, stalkwise condition.
\end{remark}

\begin{proposition}
\label{ringed-spaces-proposition-sheaves-abelian}
For a topological space $X$, the category of
sheaves of vector spaces on $X$ is abelian.
Kernels and cokernels are determined stalkwise.
\end{proposition}

\begin{definition}
\label{definition-exact-stalkwise}
A sequence of sheaves is exact if the induced
sequence on every stalk is exact.
\end{definition}

\begin{proposition}
\label{proposition-exact-stalks}
Exactness of sheaves can be checked on stalks.
\end{proposition}

\section{Restriction and local systems}
\label{ringed-spaces-section-restriction-local-systems}

\begin{definition}
\label{definition-restriction-sheaf}
Let $\mathcal F$ be a sheaf on $X$ and
$U\subset X$ open.  The restriction sheaf
$\mathcal F|_U$ is the sheaf on $U$ defined by
$$
(\mathcal F|_U)(V)=\mathcal F(V)
$$
for every open $V\subset U$.
\end{definition}

\begin{definition}
\label{definition-local-system}
Let $M$ be a finite-dimensional vector space.
A sheaf $\mathcal F$ on $X$ is a {\it local system}
with fibre $M$ if every $p\in X$ has a
neighbourhood $U$ for which
$$
\mathcal F|_U\cong \underline M_U.
$$
\end{definition}

\begin{example}[Locally constant but not constant]
\label{example-local-system-not-constant}
On $X=\mathbb C^\times$, the local branches of
$z^{2/3}$ span a rank-one local system.  On a
small ball avoiding the origin a branch can be
chosen, so the restriction is constant; going
around the origin changes the branch, so the
local system is not globally constant.
\end{example}

\begin{proposition}[Sheaves on a basis]
\label{ringed-spaces-proposition-sheaf-on-basis}
Let $\mathcal B$ be a basis for the topology of
$X$.  Restricting a sheaf on $X$ to the opens in
$\mathcal B$ gives an equivalence between sheaves
on $X$ and sheaves specified on the basis
$\mathcal B$.
\end{proposition}

\section{Direct and inverse image of sheaves}
\label{ringed-spaces-section-image-functors}
\label{section-pushforward}
\label{section-pullback}

\begin{definition}
\label{definition-pushforward}
Let $f:X\to Y$ be continuous and let
$\mathcal F$ be a sheaf on $X$.  Its
{\it direct image} or {\it pushforward} is the
sheaf $f_*\mathcal F$ on $Y$ given by
$$
(f_*\mathcal F)(V)
=
\mathcal F(f^{-1}(V)).
$$
\end{definition}

\begin{remark}
\label{remark-pushforward-sheaf}
The pushforward is a genuine sheaf without any
additional sheafification.
\end{remark}

\begin{proposition}
\label{ringed-spaces-proposition-pushforward-left-exact}
The functor $f_*$ is left exact.
\end{proposition}

\begin{definition}
\label{definition-pullback}
Let $\mathcal G$ be a sheaf on $Y$.  Define a
presheaf on $X$ by
$$
U
\longmapsto
\varinjlim_{V\supset f(U)}
\mathcal G(V),
$$
where $V$ runs over open subsets of $Y$ containing
$f(U)$.  Its sheafification is the
{\it inverse-image sheaf} $f^{-1}\mathcal G$.
\end{definition}

\begin{proposition}
\label{ringed-spaces-proposition-inverse-image-stalk}
The inverse-image sheaf is characterized on stalks
by
$$
(f^{-1}\mathcal G)_x
\cong
\mathcal G_{f(x)}.
$$
In particular, the functor $f^{-1}$ is exact.
\end{proposition}

\begin{proposition}[Adjunction]
\label{ringed-spaces-proposition-inverse-image-pushforward-adjunction}
The functors $f^{-1}$ and $f_*$ form an adjoint
pair:
$$
\operatorname{Hom}(f^{-1}\mathcal G,\mathcal F)
\cong
\operatorname{Hom}(\mathcal G,f_*\mathcal F),
$$
naturally in $\mathcal F$ and $\mathcal G$.
\end{proposition}

\begin{remark}
If $\pi:X\to *$ is the map to a point, then
$$
\Gamma(X,-)=\pi_*(-).
$$
Thus the global-sections functor is a special case
of pushforward and is left exact.
\end{remark}

\section{The sheaf Hom}
\label{ringed-spaces-section-sheaf-hom}

\noindent
The naive assignment
$U\mapsto\operatorname{Hom}(\mathcal F(U),
\mathcal G(U))$ does not naturally restrict.
The correct local Hom uses restriction sheaves.

\begin{definition}
\label{ringed-spaces-definition-sheaf-hom}
For sheaves $\mathcal F,\mathcal G$ on $X$, define
$$
\SheafHom(\mathcal F,\mathcal G)(U)
=
\operatorname{Hom}
(\mathcal F|_U,\mathcal G|_U).
$$
This is a sheaf on $X$.
\end{definition}

\noindent
In particular,
$$
\Gamma\bigl(X,\SheafHom(\mathcal F,\mathcal G)\bigr)
=
\operatorname{Hom}(\mathcal F,\mathcal G).
$$

\section{Sheaves of \texorpdfstring{$\mathcal O$}{O}-modules}
\label{ringed-spaces-section-o-modules}
\label{section-rings-modules}
\label{section-ringed-spaces}
\label{section-locally-ringed}

\begin{definition}
\label{definition-ringed-space}
A {\it ringed space} is a pair
$(X,\mathcal O_X)$ consisting of a topological
space and a sheaf of rings on it.
\end{definition}

\begin{definition}
\label{definition-sheaf-ring}
An $\mathcal O_X$-module is a sheaf
$\mathcal F$ on $X$ such that every
$\mathcal F(U)$ is an $\mathcal O_X(U)$-module,
compatibly with restriction maps.
\end{definition}

\begin{definition}
\label{definition-locally-ringed}
A ringed space $(X,\mathcal O_X)$ is
{\it locally ringed} if each stalk
$\mathcal O_{X,p}$ is a local ring.
\end{definition}

\begin{definition}
\label{definition-lrs-morphism}
A morphism of locally ringed spaces
$f:(X,\mathcal O_X)\to(Y,\mathcal O_Y)$ consists
of a continuous map together with a morphism
$$
f^{-1}\mathcal O_Y\longrightarrow\mathcal O_X
$$
whose induced maps on local rings are local
homomorphisms.
\end{definition}

\subsection{Pushforward and pullback of modules}

\noindent
If $\mathcal F$ is an $\mathcal O_X$-module, then
$f_*\mathcal F$ is naturally an
$\mathcal O_Y$-module through
$\mathcal O_Y\to f_*\mathcal O_X$.  At the level
of underlying sheaves this is the ordinary
pushforward, so $f_*$ is left exact.

If $\mathcal G$ is an $\mathcal O_Y$-module, the
ordinary inverse image $f^{-1}\mathcal G$ is only
an $f^{-1}\mathcal O_Y$-module.  The module
pullback is therefore defined by
$$
f^*\mathcal G
=
\mathcal O_X
\otimes_{f^{-1}\mathcal O_Y}
f^{-1}\mathcal G.
$$

\begin{proposition}
\label{ringed-spaces-proposition-module-pullback}
The functors $f^*$ and $f_*$ on module sheaves
are adjoint.  In general $f^*$ is right exact,
not exact.
\end{proposition}

\begin{definition}
\label{ringed-spaces-definition-flat-morphism}
A morphism of ringed spaces is {\it flat} if
$f^*$ is exact.
\end{definition}

\subsection{Affine schemes and quasi-coherent sheaves}

\noindent
Let $R$ be a commutative ring.  The spectrum
$$
\operatorname{Spec}R
=
\{\mathfrak p\subset R:\mathfrak p
\text{ prime}\}
$$
carries the Zariski topology.  The distinguished
opens
$$
D(f)
=
\operatorname{Spec}R\setminus V(f)
$$
form a basis.  The structure sheaf is determined
on this basis by
$$
\mathcal O_{\operatorname{Spec}R}(D(f))
=
R_f.
$$
Likewise an $R$-module $M$ determines a sheaf
$\widetilde M$ with
$$
\widetilde M(D(f))=M_f.
$$

\begin{definition}
\label{ringed-spaces-definition-quasi-coherent}
An $\mathcal O_X$-module $\mathcal F$ on a scheme
$X$ is {\it quasi-coherent} if $X$ admits an affine
open cover $U_i$ such that
$$
\mathcal F|_{U_i}\cong\widetilde{M_i}
$$
for some $\mathcal O_X(U_i)$-module $M_i$.
\end{definition}


\section{Glueing ringed spaces}
\label{section-glueing-ringed-spaces}

\noindent
Let's start with Hartshorne's exercise,
then I'll paste the Stacks Project version.

An interesting special case is when the
family $X_i$ is arbitrary, but the $U_{ij}$
and $\varphi_{ij}$ are all empty.
Then the scheme $X$ is called the
{\it disjoint union} of the $X_i$,
and is denoted $\coprod X_i$.

\begin{exercise}
\label{exercise-gluing-schemes}
\begin{reference}
\cite[Exercise II.2.12]{hart}.
\end{reference}
Let $\{X_i\}$ be a family of schemes
(possibly infinite).
For each $i \neq j$,
suppose given an open subset $U_{ij}
\subset X_i$,
and let it have the induced scheme
structure. Suppose also given for each
$i \neq j$ an isomorphism of schemes
$\varphi_{ij}:U_{ij}\to U_{ji}$
such that (1) for each $i,j$,
$\varphi_{ji}=\varphi_{ij}^{-1}$,
and (2) for each $i,j,k$,
$\varphi_{ij}(U_{ij}\cap U_{ik}
=U_{ji}\cap U_{jk}$ and
$\varphi_{ik}= \varphi_{jk}\circ \varphi_{ij}$
on $U_{ij}\cap U_{ik}$.
Then show that there is a scheme $X$,
together with morphisms
$\psi_i:X_i \to X$ for each $i$,
such that (1) $\psi_i$ is an isomorphism
of $X_i$ onto an open subsceme of $X$,
(2) the $\psi_i(X_i)$ cover $X$,
(3) $\psi_i(U_{ij}) = \psi_i(X_i)
\cap\psi_j(X_j)$ and
(4) $\psi_i = \psi_j \circ \varphi_{ij}$
on $U_{ij}$.
\end{exercise}

\begin{proof}
Step 1. Define the underlying topological 
space:
$$
X = (\coprod_{i} X_i)/\sim
$$
where $x_i \in X_i$ and $x_j \in X_j$
are identified if $x_i \in U_{ij}$,
$x_j \in U_{ji}$ and $\varphi_{ij}(x_i)=x_j$.
Equip this space with the quotient topology.
This defines maps of topological spaces
(i.e. continuous maps)
$$
\psi_i: X_i \to X,\qquad 
x_i \mapsto [x_i]
$$
which are just the inclusions followed
by the quotient projections.
To prove $\psi_i$ is an embedding
we must show it's a homeomorphism, which
follows from it being bijective and open.
For bijective we only need to prove
injectivity. It might look immediate since
$X_i \hookrightarrow \coprod_j X_j$ is an
embedding, and projecting would only glue
points of $X_i$ with points of other $X_j$.
The nuance is that there could be
``zigzags'': $x_i$ from $X_i$ glues to
$x_j \in X_j$, which glues to $x_k \in X_k$
and that one back to $x_i' \in X_i$,
breaking injectivity.

To fix this let's be more precise:
$x_i$ being glued to $x_j$ means that
$\varphi_{ij}(x_i) = x_j$.
Similarly $\varphi_{j k}(x_j) = x_k$
and $\varphi_{ki}(x_k)=x_i'$.
But the cocycle condition says
$\varphi_{ik}=\varphi_{jk}\varphi_{ij}$,
which means that both $x_i'$ and $x_i$
map to $x_k$. But $\varphi_{ik}$ is a
homeomorphism, so it is injective, so
$x_i'=x_i$.

To conclude injectivity from this argument
suppose that $x_i, x_i' \in X_i$
are such that $\psi_i(x_i) = \psi_i(x_i')$.
This means that there is a zigzag as above,
involving possibly more than three maps,
since no two points of $X_i$ are
glued directly in the quotient.
Then we would just apply the cocycle
condition inductively and conclude as above.

Having proved injectivity, it still
remains to show that $\psi_i$ is an open map.
So we pick an open set $V$ of
$X_i$ and check that its image is open.
Since $X$ has the quotient topology,
this means that $\psi_i(V)$ is open
if its preimage under the projection map
$q$ is open in the coproduct $\coprod X_i$.
The nuance is that such preimage
may be spread over other $X_j$, 
$j\neq i$. However, since $\varphi_{ij}$
are homeomorphisms, the union of all
the ``spread-over'' parts remains a union
of (arbitrarily many) open sets, which is
open.

Step 2. Define the structure sheaf of $X$.
Let $U$ be an open set of $X$. Define
a presheaf that maps $U$ to 
$$
\mathcal{O}_X(U)
:=
(\bigotimes \mathcal{O}_{X_i}(U \cap X_i))
/\sim
$$
(since coproduct in the category
of rings is direct sum
by Exercise
\ref{commutative-algebra-exercise-coproduct-in-rings})
where two sections $s_i \in
\mathcal{O}_{X_i}(U \cap \psi_i(X_i))$
and $s_j \in \mathcal{O}_{X_j}(U \cap
\psi_j(X_j))$ 
are identified… it's not clear yet
why we need the identification.

To show this is a sheaf suppose we have
an inclusion $U \hookrightarrow V$.
Then we have a restriction map
$\rho_{U,V}:\mathcal{O}_X(V)
\to \mathcal{O}_X(U)$
given by the restrictions of
every section obtained after projecting
to each $\mathcal{O}_{X_i}(U 
\cap \psi_i(X_i))$.
This preserves compositions
and maps the identity
morphism to the identity morphism
after projecting to each summand,
and thus also on $X$.
More explicitly: for the identity
$\text{id}:U \to U$ we obtain a map
of sections $\rho_{U,U}$ putting
$(s_i) \mapsto (\rho^{X_i}_{U \cap X_i,
U\cap X_i}s_i)$
but the latter maps are the identity
because each $\mathcal{O}_{X_i}$ is indeed
a presheaf. Composition and the 
glueing condition for sheaves are
confirmed similarly.

Step 3. The conditions (1) to (4)
from the exercise are confirmed by
construction.
\end{proof}

Let $I$ be a set.
For each $i \in I$ let $(X_i, \mathcal{O}_i)$ be
a locally ringed space. (Actually the construction that
follows works equally well for ringed spaces.)
For each pair $i, j \in I$ let $U_{ij} \subset X_i$
be an open subspace.
For each pair $i, j \in I$, let
$$
\varphi_{ij} : U_{ij} \to U_{ji}
$$
be an isomorphism of locally ringed spaces.
For convenience we assume that $U_{ii} = X_i$.
For each triple $i, j, k \in I$ assume that
\begin{enumerate}
\item we have
$\varphi_{ij}^{-1}(U_{ji} \cap U_{jk}) =  U_{ij} \cap U_{ik}$, and
\item the diagram
$$
\xymatrix{
U_{ij} \cap U_{ik} \ar[rr]_{\varphi_{ik}} \ar[rd]_{\varphi_{ij}} & &
U_{ki} \cap U_{kj} \\
& U_{ji} \cap U_{jk} \ar[ru]_{\varphi_{jk}}
}
$$
is commutative.
\end{enumerate}
(In particular, it follows that $\varphi_{ii} = \text{id}_{X_i}$.)
Let us call a collection
$(I, (X_i)_{i\in I}, (U_{ij})_{i, j\in I}, (\varphi_{ij})_{i, j\in I})$
satisfying the conditions above a glueing data.

\begin{lemma}
\label{lemma-glue}
\begin{slogan}
If you have two locally ringed spaces, and a subspace of the first one
is isomorphic to a subspace of the other, then you can glue them together
into one big locally ringed space.
\end{slogan}
Given any glueing data of locally ringed spaces there
exists a locally ringed space $X$ and open subspaces
$U_i \subset X$ together with isomorphisms
$\varphi_i : X_i \to U_i$ of locally ringed spaces such that
\begin{enumerate}
\item $X=\bigcup_{i\in I} U_i$,
\item $\varphi_i(U_{ij}) = U_i \cap U_j$, and
\item $\varphi_{ij} =
\varphi_j^{-1}|_{U_i \cap U_j} \circ \varphi_i|_{U_{ij}}$.
\end{enumerate}
The locally ringed space $X$ is characterized by the following
mapping properties: Given a locally ringed space $Y$ we have
\begin{eqnarray*}
\Mor(X, Y) & = & \{ (f_i)_{i\in I} \mid
f_i : X_i \to Y, \ f_j \circ \varphi_{ij} = f_i|_{U_{ij}}\} \\
f & \mapsto & (f|_{U_i} \circ \varphi_i)_{i \in I} \\
\Mor(Y, X) & = &
\left\{
\begin{matrix}
\text{open covering }Y = \bigcup\nolimits_{i \in I} V_i\text{ and }
(g_i : V_i \to X_i)_{i \in I}
\text{ such that}\\
g_i^{-1}(U_{ij}) = V_i \cap V_j
\text{ and }
g_j|_{V_i \cap V_j} = \varphi_{ij} \circ g_i|_{V_i \cap V_j}
\end{matrix}
\right\} \\
g & \mapsto &
V_i = g^{-1}(U_i), \ g_i = \varphi_i^{-1} \circ g|_{V_i}
\end{eqnarray*}
\end{lemma}

\begin{proof}
We construct $X$ in stages.
As a set we take
$$
X = (\coprod X_i) / \sim.
$$
Here given $x \in X_i$ and $x' \in X_j$ we say
$x \sim x'$ if and only if $x \in U_{ij}$, $x' \in U_{ji}$
and $\varphi_{ij}(x) = x'$. This is an equivalence relation
since if $x \in X_i$, $x' \in X_j$, $x'' \in X_k$, and $x \sim x'$ and
$x' \sim x''$, then $x' \in U_{ji} \cap U_{jk}$, hence by condition (1) of
a glueing data also $x \in U_{ij} \cap U_{ik}$ and
$x'' \in U_{ki} \cap U_{kj}$ and by condition (2)
we see that $\varphi_{ik}(x) = x''$. (Reflexivity and symmetry
follows from our assumptions that $U_{ii} = X_i$ and
$\varphi_{ii} = \text{id}_{X_i}$.)
Denote $\varphi_i : X_i \to X$
the natural maps. Denote $U_i = \varphi_i(X_i) \subset X$.
Note that $\varphi_i : X_i \to U_i$ is a bijection.

\medskip\noindent
The topology on $X$ is defined by the rule that
$U \subset X$ is open if and only if $\varphi_i^{-1}(U)$
is open for all $i$. We leave it to the reader to verify
that this does indeed define a topology.
Note that in particular $U_i$ is open since $\varphi_j^{-1}(U_i)
= U_{ji}$ which is open in $X_j$ for all $j$.
Moreover, for any open set $W \subset X_i$ the image
$\varphi_i(W) \subset U_i$ is open because
$\varphi_j^{-1}(\varphi_i(W)) = \varphi_{ji}^{-1}(W \cap U_{ij})$.
Therefore $\varphi_i : X_i \to U_i$ is a homeomorphism.

\medskip\noindent
To obtain a locally ringed space we have to construct the
sheaf of rings $\mathcal{O}_X$. We do this by glueing the
sheaves of rings $\mathcal{O}_{U_i} := \varphi_{i, *} \mathcal{O}_i$.
Namely, in the commutative diagram
$$
\xymatrix{
U_{ij} \ar[rr]_{\varphi_{ij}} \ar[rd]_{\varphi_i|_{U_{ij}}}
& &
U_{ji} \ar[ld]^{\varphi_j|_{U_{ji}}} \\
& U_i \cap U_j &
}
$$
the arrow on top is an isomorphism of ringed spaces,
and hence we get unique isomorphisms of sheaves of rings
$$
\mathcal{O}_{U_i}|_{U_i \cap U_j}
\longrightarrow
\mathcal{O}_{U_j}|_{U_i \cap U_j}.
$$
These satisfy a cocycle condition as in Sheaves,
Section \ref{sheaves-section-glueing-sheaves}.
By the results of that section we obtain a sheaf of rings
$\mathcal{O}_X$ on $X$ such that $\mathcal{O}_X|_{U_i}$
is isomorphic to $\mathcal{O}_{U_i}$ compatibly with
the glueing maps displayed above.
In particular $(X, \mathcal{O}_X)$ is a locally ringed
space since the stalks of $\mathcal{O}_X$ are equal
to the stalks of $\mathcal{O}_i$ at corresponding
points.

\medskip\noindent
The proof of the mapping properties is omitted.
\end{proof}

\begin{lemma}
\label{lemma-glue-schemes}
\begin{slogan}
Schemes can be glued to give new schemes.
\end{slogan}
In Lemma \ref{lemma-glue} above, assume that all
$X_i$ are schemes. Then the resulting locally ringed
space $X$ is a scheme.
\end{lemma}

\begin{proof}
This is clear since each of the $U_i$ is a scheme
and hence every $x \in X$ has an affine neighbourhood.
\end{proof}

$X$ via the isomorphisms $\varphi_i$. We will do this in

\section{The sheaf of singular cochains}
\label{section-sheaf-singular-cochains}

\noindent
This is the construction used later in
\cite[Section 6.2.7]{vanEkerenHomologicalAlgebra2026}
to compare derived global sections of the constant
sheaf with singular cohomology.  We write out the
exactness argument explicitly.

For an open set $U\subset X$, let
$$
C^p_{\mathrm{sing}}(U;\mathbb Z)
:=
\operatorname{Hom}_{\mathbb Z}
\bigl(C_p^{\mathrm{sing}}(U),\mathbb Z\bigr).
$$
If $V\subset U$, the inclusion
$C_p^{\mathrm{sing}}(V)\to C_p^{\mathrm{sing}}(U)$
induces a restriction map
$$
C^p_{\mathrm{sing}}(U;\mathbb Z)
\longrightarrow
C^p_{\mathrm{sing}}(V;\mathbb Z).
$$
Sheafifying this presheaf gives
$\mathcal C^p_{\mathrm{sing}}$, and the singular
coboundary gives morphisms
$$
\delta:\mathcal C^p_{\mathrm{sing}}
\longrightarrow
\mathcal C^{p+1}_{\mathrm{sing}}.
$$

\begin{proposition}
\label{proposition-singular-cochains-resolve-constant-sheaf}
Let $X$ be a smooth manifold. Then
$$
0\longrightarrow \underline{\mathbb Z}_X
\longrightarrow \mathcal C^0_{\mathrm{sing}}
\xrightarrow{\delta}\mathcal C^1_{\mathrm{sing}}
\xrightarrow{\delta}\mathcal C^2_{\mathrm{sing}}
\longrightarrow\cdots
$$
is exact. Equivalently,
$\mathcal C^\bullet_{\mathrm{sing}}$ is a
resolution of the constant sheaf
$\underline{\mathbb Z}_X$.
\end{proposition}

\begin{proof}
Exactness can be checked on stalks.  For
$x\in X$,
$$
(\mathcal C^p_{\mathrm{sing}})_x
\cong
\varinjlim_{x\in U}
C^p_{\mathrm{sing}}(U;\mathbb Z).
$$
Filtered colimits are exact in $\mathbf{Ab}$, so
$$
H^p\bigl((\mathcal C^\bullet_{\mathrm{sing}})_x\bigr)
\cong
\varinjlim_{x\in U}
H^p_{\mathrm{sing}}(U;\mathbb Z).
$$
A smooth manifold has a neighbourhood basis at
$x$ consisting of contractible open sets.  Hence
the colimit is zero for $p>0$ and is
$\mathbb Z$ for $p=0$.  The map from the
constant sheaf identifies
$\underline{\mathbb Z}_{X,x}\cong\mathbb Z$
with this degree-zero cohomology.  Thus the
augmented complex is exact on every stalk.
\end{proof}

\begin{remark}
For a manifold these singular-cochain sheaves are
flabby.  Consequently they are acyclic for global
sections, which is why this resolution computes
sheaf cohomology of the constant sheaf.
\end{remark}
